export function confetti(canvas, { colors = ['#c9a227', '#7a1f2b', '#f1e9da', '#e8a0b0'], count =

  const ctx = canvas.getContext('2d');
  const dpr = Math.min(window.devicePixelRatio || 1, 2);
  const W = (canvas.width = Math.round(canvas.clientWidth * dpr));
  const H = (canvas.height = Math.round(canvas.clientHeight * dpr));
  const parts = Array.from({ length: count }, (_, i) => {
    const a = -Math.PI / 2 + (Math.random() - 0.5) * Math.PI * 0.95;
    const v = (7 + Math.random() * 10) * dpr;
    return { x: origin.x * W, y: origin.y * H, vx: Math.cos(a) * v, vy: Math.sin(a) * v,
      w: (6 + Math.random() * 6) * dpr, h: (8 + Math.random() * 10) * dpr, r: Math.random() * 
Math.PI, vr: (Math.random() - 0.5) * 0.3, c: colors[i % colors.length] };
  });
  return new Promise((resolve) => {
    const t0 = performance.now();
    const frame = (now) => {
      const t = now - t0;
      ctx.clearRect(0, 0, W, H);
      ctx.globalAlpha = Math.max(0, 1 - t / duration);
      for (const p of parts) {
        p.vy += 0.35 * dpr; // משיכה  כוח
        p.vx *= 0.985;
        p.vy *= 0.985;
        p.x += p.vx;
        p.y += p.vy;
        p.r += p.vr;
        ctx.save();
        ctx.translate(p.x, p.y);
        ctx.rotate(p.r);
        ctx.fillStyle = p.c;
        ctx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h * Math.abs(Math.cos(p.r * 2))); // מתהפך  נייר
        ctx.restore();
      }

      else { ctx.clearRect(0, 0, W, H); resolve(); }
    };

  });
